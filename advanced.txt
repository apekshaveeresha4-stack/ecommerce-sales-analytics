SELECT
    p.product_name,
    p.category,
    ROUND(SUM(o.sales), 2) AS total_sales,
    ROUND(SUM(o.profit), 2) AS total_profit,
    ROUND(SUM(o.profit) / SUM(o.sales) * 100, 2) AS profit_margin
FROM orders o
JOIN products p
    ON o.product_id = p.product_id
GROUP BY p.product_id, p.product_name, p.category
HAVING SUM(o.sales) > 500000
ORDER BY profit_margin ASC;

-- KEY INSIGHTS
-- Accessories Product 025 has the lowest profit margin among products
-- generating more than ₹500,000 in sales.
-- Accessories Product 051 also has a relatively low profit margin.
-- Several Home Appliances products generate high sales but moderate margins.
-- Fragrance Product 034 combines high sales with a strong profit margin.
-- High revenue alone does not indicate high profitability.

-- KEY INSIGHTS
-- 133 customers placed more than one order.
-- Approximately 25.3% of customers are repeat customers.
-- The majority of customers placed only one order.
-- This suggests an opportunity to improve customer retention and repeat purchases.
